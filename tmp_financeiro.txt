8:	export function parseCompetencia(competencia: string) {
9:	  // aceita 'YYYY-MM' ou datas que comecem com esse formato
10:	  const base = competencia?.slice(0, 7)
11:	  const comp = parse(base, 'yyyy-MM', new Date())
12:	  if (isNaN(comp.getTime())) {
13:	    throw new Error('Competência inválida. Use YYYY-MM')
14:	  }
15:	  return {
16:	    competenciaStr: base,
17:	    inicio: startOfMonth(comp),
18:	    fim: endOfMonth(comp),
19:	  }
20:	}

22:	export async function gerarMensalidades(
23:	  predioId: string,
24:	  competencia: string,
25:	  opts?: { vencimentoDia?: number; valorPadrao?: number; sobrescrever?: boolean }
26:	) {
27:	  const { inicio, fim, competenciaStr } = parseCompetencia(competencia)
28:	  const dia = opts?.vencimentoDia && opts.vencimentoDia >= 1 && opts.vencimentoDia <= 28 ? opts.vencimentoDia : 10
29:	  const vencimento = new Date(inicio.getFullYear(), inicio.getMonth(), dia)
30:	
31:	  const unidades = await prisma.unidade.findMany({
32:	    where: { predioId, ativo: true },
33:	    select: { id: true, valorTaxa: true },
34:	  })
35:	
36:	  let created = 0
37:	  let updated = 0
38:	  let skipped = 0
39:	
40:	  for (const u of unidades) {
41:	    const valorUnidade = Number(u.valorTaxa ?? 0)
42:	    const valor = valorUnidade > 0 ? valorUnidade : Number(opts?.valorPadrao ?? 0)
43:	    if (!valor || valor <= 0) { skipped++; continue }
44:	
45:	    const existing = await prisma.pagamento.findFirst({
46:	      where: {
47:	        predioId,
48:	        unidadeId: u.id,
49:	        tipo: PagamentoTipo.TAXA_MENSAL,
50:	        competencia: { gte: inicio, lte: fim },
51:	      },
52:	      select: { id: true },
53:	    })
54:	
55:	    if (existing) {
56:	      if (opts?.sobrescrever) {
57:	        await prisma.pagamento.update({
58:	          where: { id: existing.id },
59:	          data: { valor, vencimento, status: PagamentoStatus.PENDENTE },
60:	        })
61:	        updated++
62:	      } else {
63:	        skipped++
64:	      }
65:	    } else {
66:	      await prisma.pagamento.create({
67:	        data: {
68:	          predioId,
69:	          unidadeId: u.id,
70:	          tipo: PagamentoTipo.TAXA_MENSAL,
71:	          status: PagamentoStatus.PENDENTE,
72:	          competencia: inicio,
73:	          vencimento,
74:	          valor,
75:	        },
76:	      })
77:	      created++
78:	    }
79:	  }
80:	
81:	  return { competencia: competenciaStr, resumo: { created, updated, skipped, totalUnidades: unidades.length } }
82:	}

84:	export async function recalcularStatus(predioId: string, competencia: string) {
85:	  const { inicio, fim } = parseCompetencia(competencia)
86:	  const agora = new Date()
87:	
88:	  const whereBase: Prisma.PagamentoWhereInput = {
89:	    predioId,
90:	    competencia: { gte: inicio, lte: fim },
91:	    NOT: { status: PagamentoStatus.PAGO },
92:	  }
93:	
94:	  const atrasados = await prisma.pagamento.updateMany({
95:	    where: { ...whereBase, vencimento: { lt: agora } },
96:	    data: { status: PagamentoStatus.ATRASADO },
97:	  })
98:	
99:	  const pendentes = await prisma.pagamento.updateMany({
100:	    where: { ...whereBase, vencimento: { gte: agora } },
101:	    data: { status: PagamentoStatus.PENDENTE },
102:	  })
103:	
104:	  const counts = await prisma.pagamento.groupBy({
105:	    by: ['status'],
106:	    where: { predioId, competencia: { gte: inicio, lte: fim } },
107:	    _count: { _all: true },
108:	  })
109:	
110:	  return { updated: { atrasados: atrasados.count, pendentes: pendentes.count }, counts }
111:	}

113:	export async function getResumoV2(predioId: string, competencia: string) {
114:	  const { inicio, fim, competenciaStr } = parseCompetencia(competencia)
115:	
116:	  const pagamentos = await prisma.pagamento.findMany({
117:	    where: {
118:	      unidade: { predioId },
119:	      competencia: { gte: inicio, lte: fim },
120:	    },
121:	    select: {
122:	      id: true,
123:	      unidadeId: true,
124:	      valor: true,
125:	      status: true,
126:	      vencimento: true,
127:	      dataPagamento: true,
128:	    },
129:	  })
130:	
131:	  const MS_DIA = 86_400_000
132:	  const hoje = new Date()
133:	
134:	  const linhas = pagamentos.map((p) => {
135:	    const venc = new Date(p.vencimento)
136:	    const vencido = p.status !== 'PAGO' && venc < hoje
137:	    const statusUI: FinanceiroStatusUI = p.status === 'PAGO' ? 'PAGO' : vencido ? 'ATRASADO' : 'PENDENTE'
138:	    const diasAtraso = vencido ? Math.max(0, Math.floor((+hoje - +venc) / MS_DIA)) : 0
139:	    return { id: p.id, unidadeId: p.unidadeId, valor: Number(p.valor), statusUI, diasAtraso }
140:	  })
141:	
142:	  const totalDevido = pagamentos.reduce((acc, p) => acc + Number(p.valor), 0)
143:	  const totalRecebido = pagamentos.filter((p) => p.status === 'PAGO').reduce((acc, p) => acc + Number(p.valor), 0)
144:	
145:	  const todasUnidades = new Set(pagamentos.map((p) => p.unidadeId))
146:	  const inadimplentes = new Set(linhas.filter((l) => l.statusUI === 'ATRASADO').map((l) => l.unidadeId))
147:	  const inadimplenciaPct = todasUnidades.size === 0 ? 0 : (inadimplentes.size / todasUnidades.size) * 100
148:	
149:	  const pendencias = linhas
150:	    .filter((l) => l.statusUI !== 'PAGO')
151:	    .map(({ id, unidadeId, valor, statusUI, diasAtraso }) => ({ id, unidadeId, valor, status: statusUI, diasAtraso }))
152:	
153:	  // Despesas fixas ativas na competência
154:	  const repoDF = (prisma as any).despesaFixa as { findMany?: Function } | undefined
155:	  const despesasFixas = repoDF?.findMany
156:	    ? await repoDF.findMany({
157:	        where: {
158:	          predioId,
159:	          ativo: true,
160:	          AND: [
161:	            { inicio: { lte: fim } },
162:	            { OR: [{ fim: null }, { fim: { gte: inicio } }] },
163:	          ],
164:	        },
165:	        select: { valor: true },
166:	      })
167:	    : await prisma.$queryRaw<Array<{ valor: any }>>`
168:	        SELECT "valor"
169:	        FROM "DespesaFixa"
170:	        WHERE "predioId" = ${predioId}
171:	          AND "ativo" = true
172:	          AND ("competenciaInicio" <= ${fim}
173:	               AND ("competenciaFim" IS NULL OR "competenciaFim" >= ${inicio}))
174:	      `
175:	  const despesasFixasMes = (despesasFixas as Array<{ valor: Prisma.Decimal | number }>).reduce((acc: number, d: { valor: Prisma.Decimal | number }) => acc + Number(d.valor), 0)
176:	
177:	  return {
178:	    competencia: competenciaStr,
179:	    kpis: {
180:	      totalDevido: Number(totalDevido.toFixed(2)),
181:	      totalRecebido: Number(totalRecebido.toFixed(2)),
182:	      inadimplentes: inadimplentes.size,
183:	      inadimplenciaPct: Number(inadimplenciaPct.toFixed(2)),
184:	      despesasFixasMes: Number(despesasFixasMes.toFixed(2)),
185:	      saldoPrevisto: Number((totalRecebido - despesasFixasMes).toFixed(2)),
186:	    },
187:	    pendencias,
188:	  }
189:	}

191:	export async function listPagamentos(
192:	  predioId: string,
193:	  competencia: string,
194:	  status: 'PENDENCIAS' | 'PENDENTE' | 'ATRASADO' | 'PAGO' | 'TODOS'
195:	) {
196:	  const { inicio, fim } = parseCompetencia(competencia)
197:	
198:	  const hoje = new Date()
199:	
200:	  const rows = await prisma.pagamento.findMany({
201:	    where: {
202:	      predioId,
203:	      competencia: { gte: inicio, lte: fim },
204:	    },
205:	    orderBy: [{ vencimento: 'asc' }, { createdAt: 'asc' }],
206:	    include: { unidade: { select: { numero: true } } },
207:	  })
208:	
209:	  const itensAll = rows.map((r) => {
210:	    const vencido = r.status !== 'PAGO' && r.vencimento < hoje
211:	    const statusUI: FinanceiroStatusUI = r.status === 'PAGO' ? 'PAGO' : vencido ? 'ATRASADO' : 'PENDENTE'
212:	    return {
213:	      id: r.id,
214:	      unidadeId: r.unidadeId,
215:	      unidadeNome: r.unidade?.numero ? `Unidade ${r.unidade.numero}` : 'Condomínio',
216:	      valor: Number(r.valor),
217:	      status: statusUI,
218:	      diasAtraso: vencido ? Math.max(0, Math.floor((+hoje - +r.vencimento) / 86_400_000)) : 0,
219:	    }
220:	  })
221:	
222:	  const itens = ((): typeof itensAll => {
223:	    switch (status) {
224:	      case 'PENDENCIAS':
225:	        return itensAll.filter((i) => i.status === 'PENDENTE' || i.status === 'ATRASADO')
226:	      case 'PENDENTE':
227:	      case 'ATRASADO':
228:	      case 'PAGO':
229:	        return itensAll.filter((i) => i.status === status)
230:	      case 'TODOS':
231:	      default:
232:	        return itensAll
233:	    }
234:	  })()
235:	
236:	  return { itens }
237:	}

239:	export async function gerarLancamentosDespesasFixas(
240:	  predioId: string,
241:	  competencia: string,
242:	  opts?: { sobrescrever?: boolean }
243:	) {
244:	  const { inicio } = parseCompetencia(competencia)
245:	  // despesas ativas na competência
246:	  const repoDF2 = (prisma as any).despesaFixa as { findMany?: Function } | undefined
247:	  const despesas = repoDF2?.findMany
248:	    ? await repoDF2.findMany({
249:	        where: {
250:	          predioId,
251:	          ativo: true,
252:	          AND: [
253:	            { OR: [{ dataInicio: null }, { dataInicio: { lte: inicio } }] },
254:	            { OR: [{ dataFim: null }, { dataFim: { gte: inicio } }] },
255:	          ],
256:	        },
257:	        select: { id: true, valor: true, diaVencimento: true },
258:	      })
259:	    : await prisma.$queryRaw<Array<{ id: string; valor: any; diaVencimento: number }>>`
260:	        SELECT "id", "valor", "diaVencimento"
261:	        FROM "DespesaFixa"
262:	        WHERE "predioId" = ${predioId}
263:	          AND "ativo" = true
264:	          AND (("dataInicio" IS NULL OR "dataInicio" <= ${inicio})
265:	               AND ("dataFim" IS NULL OR "dataFim" >= ${inicio}))
266:	      `
267:	
268:	  let created = 0
269:	  let updated = 0
270:	  let skipped = 0
271:	
272:	  for (const d of despesas) {
273:	    const venc = new Date(inicio.getFullYear(), inicio.getMonth(), d.diaVencimento)
274:	    const existing = await prisma.pagamento.findFirst({
275:	      where: { predioId, despesaFixaId: d.id, competencia: { gte: inicio, lte: endOfMonth(inicio) } } as any,
276:	      select: { id: true },
277:	    })
278:	    if (existing) {
279:	      if (opts?.sobrescrever) {
280:	        await prisma.pagamento.update({ where: { id: existing.id }, data: {
281:	          valor: d.valor, vencimento: venc, status: PagamentoStatus.PENDENTE,
282:	        }})
283:	        updated++
284:	      } else {
285:	        skipped++
286:	      }
287:	    } else {
288:	      await prisma.pagamento.create({ data: ({
289:	        predioId,
290:	        // unidadeId intencionalmente omitido
291:	        despesaFixaId: d.id as any,
292:	        tipo: 'DESPESA_FIXA' as any,
293:	        status: PagamentoStatus.PENDENTE,
294:	        competencia: inicio,
295:	        vencimento: venc,
296:	        valor: d.valor as any,
297:	      } as any) })
298:	      created++
299:	    }
300:	  }
301:	
302:	  return { created, updated, skipped, totalDespesas: despesas.length }
303:	}

376:	): Promise<DespesaFixaDTO> {

385:

387:	    const rows = await prisma.$queryRaw<Array<any>>`

413:	  return {

415:	    predioId: row.predioId,
416:	    nome: row.nome,
417:	    valor: Number(row.valor),

420:	    dataInicio: row.dataInicio ?? null,

424:	}
